cmake_minimum_required(VERSION 3.28)
project(silk_consumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(SILK_SOURCE)
    add_subdirectory("${SILK_SOURCE}" silk-build)
    if(SL_BUILD_TESTS OR SL_INSTALL OR SL_SANDBOX OR SL_BUILD_BENCH OR SL_BUILD_EXAMPLE)
        message(FATAL_ERROR "Embedding unexpectedly enabled Silk developer/install targets")
    endif()
else()
    find_package(silk ${SILK_REQUESTED_VERSION} CONFIG REQUIRED)
endif()

add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE silk::silk)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/consumer-options-$<CONFIG>.txt"
    CONTENT "$<TARGET_PROPERTY:consumer,COMPILE_OPTIONS>")

# Generate one translation unit per public header, never prepend another
# header that might accidentally supply a missing dependency.
file(GLOB public_headers "${SILK_PUBLIC_HEADERS}/silk/*.h")
if(NOT public_headers)
    message(FATAL_ERROR "No public headers to verify")
endif()
foreach(header IN LISTS public_headers)
    get_filename_component(name "${header}" NAME_WE)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header_${name}.c"
        "#include <silk/${name}.h>\nint main(void) { return 0; }\n")
    add_library(header_${name} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header_${name}.c")
    target_link_libraries(header_${name} PRIVATE silk::silk)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(header_${name} PRIVATE -pedantic-errors)
    endif()
endforeach()
add_executable(linkage linkage.cpp)
target_link_libraries(linkage PRIVATE silk::silk)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(linkage PRIVATE -pedantic-errors)
endif()
